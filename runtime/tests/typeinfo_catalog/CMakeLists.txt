# Native product-path regression; also consumable against a normal installed SO.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.16)
    project(typeinfo_catalog LANGUAGES CXX)
    get_filename_component(TYPEINFO_RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
else()
    set(TYPEINFO_RUNTIME_ROOT "${CMAKE_SOURCE_DIR}")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(cj_typeinfo_catalog_unit typeinfo_catalog.cpp)
    target_compile_features(cj_typeinfo_catalog_unit PRIVATE cxx_std_17)
    target_compile_options(cj_typeinfo_catalog_unit PRIVATE -fno-rtti -fvisibility-inlines-hidden)
    target_include_directories(cj_typeinfo_catalog_unit PRIVATE
        "${TYPEINFO_RUNTIME_ROOT}/src" "${TYPEINFO_RUNTIME_ROOT}/src/Heap"
        "${TYPEINFO_RUNTIME_ROOT}/src/Loader/BinaryFile"
        "${TYPEINFO_RUNTIME_ROOT}/include"
        "${TYPEINFO_RUNTIME_ROOT}/third_party/third_party_bounds_checking_function/include")
    if(TARGET cangjie-runtime)
        set(_typeinfo_test_libraries "$<TARGET_FILE_DIR:cangjie-runtime>:$<TARGET_FILE_DIR:boundscheck>")
        add_dependencies(cj_typeinfo_catalog_unit cangjie-runtime boundscheck)
        target_link_libraries(cj_typeinfo_catalog_unit PRIVATE
            "$<TARGET_FILE:cangjie-runtime>" "$<TARGET_FILE:boundscheck>" pthread dl)
    else()
        if(NOT EXISTS "${RUNTIME_LIB_DIR}/libcangjie-runtime.so" OR NOT EXISTS "${RUNTIME_OUTPUT_ROOT}/include")
            message(FATAL_ERROR "Supply RUNTIME_LIB_DIR and its RUNTIME_OUTPUT_ROOT with generated headers")
        endif()
        set(_typeinfo_test_libraries "${RUNTIME_LIB_DIR}")
        target_include_directories(cj_typeinfo_catalog_unit PRIVATE "${RUNTIME_OUTPUT_ROOT}/include")
        target_link_libraries(cj_typeinfo_catalog_unit PRIVATE
            "${RUNTIME_LIB_DIR}/libcangjie-runtime.so" "${RUNTIME_LIB_DIR}/libboundscheck.so" pthread dl)
    endif()
    add_library(cj_typeinfo_load_image SHARED load_image.cpp)
    target_compile_features(cj_typeinfo_load_image PRIVATE cxx_std_17)
    target_compile_options(cj_typeinfo_load_image PRIVATE -fno-rtti)
    get_target_property(_typeinfo_includes cj_typeinfo_catalog_unit INCLUDE_DIRECTORIES)
    target_include_directories(cj_typeinfo_load_image PRIVATE ${_typeinfo_includes})
    # LINK_LIBRARIES also contains the parent's whole-archive trace library.
    # Each DSO already inherits that directory setting; copy only product SOs.
    if(TARGET cangjie-runtime)
        set(_typeinfo_links "$<TARGET_FILE:cangjie-runtime>" "$<TARGET_FILE:boundscheck>" pthread dl)
    else()
        set(_typeinfo_links "${RUNTIME_LIB_DIR}/libcangjie-runtime.so" "${RUNTIME_LIB_DIR}/libboundscheck.so" pthread dl)
    endif()
    target_link_libraries(cj_typeinfo_load_image PRIVATE ${_typeinfo_links})
    add_library(cj_typeinfo_enum_image SHARED load_image.cpp)
    target_compile_features(cj_typeinfo_enum_image PRIVATE cxx_std_17)
    target_compile_options(cj_typeinfo_enum_image PRIVATE -fno-rtti)
    target_compile_definitions(cj_typeinfo_enum_image PRIVATE TYPEINFO_LOAD_ENUM=1)
    target_include_directories(cj_typeinfo_enum_image PRIVATE ${_typeinfo_includes})
    target_link_libraries(cj_typeinfo_enum_image PRIVATE ${_typeinfo_links})
    add_dependencies(cj_typeinfo_catalog_unit cj_typeinfo_load_image cj_typeinfo_enum_image)
    enable_testing()
    add_test(NAME typeinfo_catalog_fe_enum COMMAND cj_typeinfo_catalog_unit load-enum $<TARGET_FILE:cj_typeinfo_enum_image>)
    add_test(NAME typeinfo_catalog_public_load COMMAND cj_typeinfo_catalog_unit load $<TARGET_FILE:cj_typeinfo_load_image>)
    add_test(NAME typeinfo_catalog_public_load_control COMMAND cj_typeinfo_catalog_unit load-control $<TARGET_FILE:cj_typeinfo_load_image>)
    add_test(NAME typeinfo_catalog_registration COMMAND cj_typeinfo_catalog_unit registration)
    add_test(NAME typeinfo_catalog_reentrant COMMAND cj_typeinfo_catalog_unit reentrant)
    add_test(NAME typeinfo_catalog_unload COMMAND cj_typeinfo_catalog_unit unload)
    set_tests_properties(typeinfo_catalog_fe_enum typeinfo_catalog_public_load typeinfo_catalog_public_load_control typeinfo_catalog_registration typeinfo_catalog_reentrant typeinfo_catalog_unload
        PROPERTIES TIMEOUT 15 ENVIRONMENT "LD_LIBRARY_PATH=${_typeinfo_test_libraries}")
endif()
