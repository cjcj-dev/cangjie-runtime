cmake_minimum_required(VERSION 3.16.5)
project(CangjieExportSources LANGUAGES C CXX)

# Configure, but never build, every std native directory using its real source
# selection. This fallback avoids unrelated third-party builds at configure time.
if(NOT IS_DIRECTORY "${STDLIB_SOURCE}/libs")
    message(FATAL_ERROR "STDLIB_SOURCE must name a pinned stdlib source tree")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(EXPORT_SOURCE_SYSTEM "${CMAKE_SYSTEM_NAME}" CACHE INTERNAL "Source selection target")
set(CANGJIE_CODEGEN_CJNATIVE_BACKEND ON)
string(TOLOWER "${CMAKE_SYSTEM_NAME}_${CMAKE_SYSTEM_PROCESSOR}" TARGET_TRIPLE_DIRECTORY_PREFIX)
list(APPEND CMAKE_MODULE_PATH "${STDLIB_SOURCE}/cmake/modules" "${STDLIB_SOURCE}/libs/cmake")
include(AddCangjieSource)
file(GLOB_RECURSE native_cmake "${STDLIB_SOURCE}/libs/CMakeLists.txt")
list(FILTER native_cmake INCLUDE REGEX "/native/CMakeLists[.]txt$")
if(NOT native_cmake)
    message(FATAL_ERROR "No std native CMake configurations found")
endif()
list(SORT native_cmake)
foreach(config IN LISTS native_cmake)
    get_filename_component(directory "${config}" DIRECTORY)
    file(RELATIVE_PATH relative "${STDLIB_SOURCE}/libs" "${directory}")
    message(STATUS "EXPORT_SOURCE_DIRECTORY ${relative}")
    add_subdirectory("${directory}" "native/${relative}")
endforeach()
