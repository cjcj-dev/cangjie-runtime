GC_OTHER_VM_TEST(ZVerify, RuntimeRejectsWorkerPrivateMarkStackAtStart)
{
    if (!ZVerifyMarking) {
        GC_EXPECT_EQ(setenv("ZVerifyMarking", "1", 1), 0);
        RunInOtherVm("ZVerify.RuntimeRejectsWorkerPrivateMarkStackAtStart");
        return;
    }
    ExpectSceneAbort("Thread marking stack is not empty", [&] {
        RuntimeParam param{};
        param.coParam.processorNum = 1;
        param.heapParam.heapSize = 32 * 1024;
        if (InitCJRuntime(&param) != E_OK) { _exit(121); }
        ConcurrentGCBreakpoints::AcquireControl();
        ThreadGCData* workerData = nullptr;
        Heap::GetHeap().old().Workers()->threads_do([&](WorkerThread* worker) {
            if (workerData == nullptr) { workerData = const_cast<ThreadGCData*>(worker->gc_data()); }
        });
        if (workerData == nullptr) { _exit(122); }
        auto* stack = MarkStripeStack::Create(true);
        if (stack == nullptr) { _exit(123); }
        stack->Push(MarkStackEntry(uintptr_t(0x1000), true, true, true, false));
        // The major driver is idle and its workers have no task. Alter the
        // actual worker-owned stack; the controlling mutator's saferegion
        // transition must not flush this witness into a shared stripe.
        auto& privateStacks = workerData->markStacks[1];
        privateStacks.Install(0, stack);
        if (privateStacks.Population() != 1) { _exit(124); }
        std::fprintf(stderr, "VERIFY_RUNTIME_MARK_START private_stacks=1\n");
        (void)ConcurrentGCBreakpoints::RunTo("AFTER MARKING STARTED");
        _exit(0);
    });
}
